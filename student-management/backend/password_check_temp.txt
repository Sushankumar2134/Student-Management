package main

import (
	"fmt"
	"golang.org/x/crypto/bcrypt"
)

func main() {
	hash := "$2a$10$htpixOMGbp2N1mrNEFy7oeCX0usfW5q2oPVKoHxlnbUk.nAAJQ1Py"

	err := bcrypt.CompareHashAndPassword(
		[]byte(hash),
		[]byte("Satish@123"),
	)

	if err == nil {
		fmt.Println("Password matches!")
	} else {
		fmt.Println("Password does not match!")
	}
}
